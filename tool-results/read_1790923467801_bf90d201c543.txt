     1→"use client";
     2→
     3→import * as React from "react";
     4→import { motion } from "framer-motion";
     5→import {
     6→  ArrowLeft,
     7→  Eye,
     8→  GitMerge,
     9→  MousePointerClick,
    10→  MoveHorizontal,
    11→  RotateCcw,
    12→  Ruler,
    13→  Scissors,
    14→  Sparkles,
    15→  TriangleAlert,
    16→  ZoomIn,
    17→  ZoomOut,
    18→} from "lucide-react";
    19→import { toast } from "sonner";
    20→
    21→import { Badge } from "@/components/ui/badge";
    22→import { Button } from "@/components/ui/button";
    23→import { ThemeToggle } from "@/components/theme-toggle";
    24→import { ModelsCard } from "@/components/atelier/models-card";
    25→import { MeasuresCard, type MeasureField } from "@/components/atelier/measures-card";
    26→import { ClientsCard, type ApiClient } from "@/components/atelier/clients-card";
    27→import { FabricTable } from "@/components/atelier/fabric-table";
    28→import { AssemblyPlayer } from "@/components/atelier/assembly-player";
    29→import { GarmentPreview } from "@/components/atelier/garment-preview";
    30→import {
    31→  ModelIcon,
    32→  PieceMini,
    33→  ProgressRing,
    34→} from "@/components/atelier/pieces";
    35→import {
    36→  FOLD,
    37→  LETTERS,
    38→  MODELS,
    39→  buildLayout,
    40→  meterage,
    41→  type ModelKey,
    42→} from "@/lib/atelier/patterns";
    43→
    44→const reveal = {
    45→  initial: { opacity: 0, y: 18 },
    46→  whileInView: { opacity: 1, y: 0 },
    47→  viewport: { once: true, margin: "-30px" },
    48→  transition: { duration: 0.5, ease: "easeOut" as const },
    49→};
    50→
    51→const clampN = (v: string, d: number, a: number, b: number) => {
    52→  const n = +v;
    53→  return Number.isFinite(n) ? Math.min(b, Math.max(a, n)) : d;
    54→};
    55→
    56→export function AtelierApp({
    57→  initialModel,
    58→  onHome,
    59→}: {
    60→  initialModel?: ModelKey;
    61→  onHome: () => void;
    62→}) {
    63→  /* ----------------------------- état ----------------------------- */
    64→  const [modelKey, setModelKey] = React.useState<ModelKey>(initialModel ?? "droite");
    65→  const [P, setP] = React.useState("90");
    66→  const [T, setT] = React.useState("70");
    67→  const [H, setH] = React.useState("98");
    68→  const [L, setL] = React.useState(String(MODELS[initialModel ?? "droite"].L));
    69→  const [W, setW] = React.useState("140");
    70→  const [S, setS] = React.useState("1.5");
    71→  const [C, setC] = React.useState("#2A9DB5");
    72→
    73→  const [zoom, setZoom] = React.useState(1);
    74→  const [cut, setCut] = React.useState<Set<number>>(new Set());
    75→  const [cuttingId, setCuttingId] = React.useState<number | null>(null);
    76→  const [asmStep, setAsmStep] = React.useState(0);
    77→
    78→  const [clients, setClients] = React.useState<ApiClient[] | null>(null);
    79→  const [clBusy, setClBusy] = React.useState(false);
    80→  const [clientSel, setClientSel] = React.useState("");
    81→  const [clientName, setClientName] = React.useState("");
    82→
    83→  const cutRef = React.useRef<Set<number>>(new Set());
    84→  const busyRef = React.useRef<number | false>(false);
    85→  const genRef = React.useRef(0);
    86→
    87→  /* --------------------------- dérivés ---------------------------- */
    88→  const model = MODELS[modelKey];
    89→  const sa = +S || 0;
    90→  const fw = +W || 140;
    91→
    92→  const mm = React.useMemo(
    93→    () => ({
    94→      P: clampN(P, 90, 60, 160),
    95→      T: clampN(T, 70, 40, 160),
    96→      H: clampN(H, 98, 60, 180),
    97→      L: clampN(L, model.L, 15, 200),
    98→    }),
    99→    [P, T, H, L, model.L]
   100→  );
   101→
   102→  const defs = React.useMemo(() => model.g(mm), [model, mm]);
   103→  const layout = React.useMemo(() => buildLayout(defs, sa, fw), [defs, sa, fw]);
   104→  const meters = meterage(layout);
   105→  const tooNarrow = layout.mx > layout.raw;
   106→  const total = layout.pieces.length;
   107→  const cutCount = cut.size;
   108→
   109→  const sig = `${modelKey}|${P}|${T}|${H}|${L}|${W}|${S}|${C}`;
   110→
   111→  /* Toute modification réinitialise la session de coupe */
   112→  React.useEffect(() => {
   113→    genRef.current += 1;
   114→    cutRef.current = new Set();
   115→    setCut(new Set());
   116→    setCuttingId(null);
   117→    busyRef.current = false;
   118→    setAsmStep(0);
   119→  }, [sig]);
   120→
   121→  /* --------------------- fiches clientes (API) --------------------- */
   122→  const refreshClients = React.useCallback(async () => {
   123→    try {
   124→      const r = await fetch("/api/clients", { cache: "no-store" });
   125→      if (!r.ok) throw new Error();
   126→      const j = (await r.json()) as { clients?: ApiClient[] };
   127→      setClients(j.clients ?? []);
   128→    } catch {
   129→      setClients([]);
   130→      toast.error("Backend injoignable : fiches clientes indisponibles.");
   131→    }
   132→  }, []);
   133→
   134→  React.useEffect(() => {
   135→    void refreshClients();
   136→  }, [refreshClients]);
   137→
   138→  const saveClient = async () => {
   139→    const nom = clientName.trim();
   140→    if (!nom) {
   141→      toast.error("Donnez un nom à la fiche avant d'enregistrer.");
   142→      return;
   143→    }
   144→    setClBusy(true);
   145→    try {
   146→      const payload = { name: nom, P: mm.P, T: mm.T, H: mm.H, L: mm.L };
   147→      const res = await fetch(
   148→        clientSel ? `/api/clients/${clientSel}` : "/api/clients",
   149→        {
   150→          method: clientSel ? "PUT" : "POST",
   151→          headers: { "Content-Type": "application/json" },
   152→          body: JSON.stringify(payload),
   153→        }
   154→      );
   155→      const j = (await res.json().catch(() => ({}))) as {
   156→        client?: ApiClient;
   157→        error?: string;
   158→      };
   159→      if (!res.ok || !j.client) throw new Error(j.error ?? "Enregistrement impossible.");
   160→      const client = j.client;
   161→      setClients((prev) =>
   162→        prev
   163→          ? clientSel
   164→            ? prev.map((c) => (c.id === client.id ? client : c))
   165→            : [...prev, client]
   166→          : [client]
   167→      );
   168→      setClientSel(client.id);
   169→      setClientName("");
   170→      toast.success(
   171→        clientSel ? `Fiche de ${nom} mise à jour.` : `Fiche de ${nom} enregistrée.`,
   172→        { description: "Mesures conservées dans la base de l'atelier." }
   173→      );
   174→    } catch (e) {
   175→      toast.error(e instanceof Error ? e.message : "Enregistrement impossible.");
   176→    } finally {
   177→      setClBusy(false);
   178→    }
   179→  };
   180→
   181→  const deleteClient = async () => {
   182→    if (!clientSel || clBusy) return;
   183→    setClBusy(true);
   184→    const c = clients?.find((x) => x.id === clientSel);
   185→    try {
   186→      const res = await fetch(`/api/clients/${clientSel}`, { method: "DELETE" });
   187→      if (!res.ok) throw new Error("Suppression impossible.");
   188→      setClients((prev) => prev?.filter((x) => x.id !== clientSel) ?? []);
   189→      setClientSel("");
   190→      if (c) toast(`Fiche de ${c.name} supprimée.`);
   191→    } catch (e) {
   192→      toast.error(e instanceof Error ? e.message : "Suppression impossible.");
   193→    } finally {
   194→      setClBusy(false);
   195→    }
   196→  };
   197→
   198→  const applyClient = (id: string) => {
   199→    setClientSel(id);
   200→    const c = clients?.find((x) => x.id === id);
   201→    if (!c) return;
   202→    setP(String(c.P));
   203→    setT(String(c.T));
   204→    setH(String(c.H));
   205→    setL(String(c.L));
   206→    setClientName(c.name);
   207→    toast(`Mesures de ${c.name} appliquées.`);
   208→  };
   209→
   210→  /* ------------------------- découpe ------------------------------ */
   211→  const runCut = (id: number) => {
   212→    const gen = genRef.current;
   213→    return new Promise<void>((resolve) => {
   214→      if (cutRef.current.has(id) || busyRef.current !== false) {
   215→        resolve();
   216→        return;
   217→      }
   218→      busyRef.current = id;
   219→      setCuttingId(id);
   220→      window.setTimeout(() => {
   221→        if (gen !== genRef.current) {
   222→          resolve();
   223→          return;
   224→        }
   225→        const next = new Set(cutRef.current);
   226→        next.add(id);
   227→        cutRef.current = next;
   228→        setCut(next);
   229→        setCuttingId(null);
   230→        busyRef.current = false;
   231→        if (total > 0 && next.size === total) {
   232→          toast.success("Toutes les pièces sont coupées !", {
   233→            description:
   234→              "Reportez les repères avant de retirer le patron, puis passez à l'assemblage.",
   235→          });
   236→        }
   237→        resolve();
   238→      }, 1050);
   239→    });
   240→  };
   241→
   242→  const cutAll = async () => {
   243→    if (busyRef.current !== false) return;
   244→    for (const p of layout.pieces) {
   245→      if (cutRef.current.has(p.id)) continue;
   246→      await runCut(p.id);
   247→    }
   248→  };
   249→
   250→  const resetCut = () => {
   251→    genRef.current += 1;
   252→    cutRef.current = new Set();
   253→    setCut(new Set());
   254→    setCuttingId(null);
   255→    busyRef.current = false;
   256→  };
   257→
   258→  /* --------------------------- divers ------------------------------ */
   259→  const changeModel = (k: ModelKey) => {
   260→    setModelKey(k);
   261→    setL(String(MODELS[k].L));
   262→  };
   263→
   264→  const setMeasure = (k: MeasureField, v: string) => {
   265→    switch (k) {
   266→      case "P":
   267→        setP(v);
   268→        break;
   269→      case "T":
   270→        setT(v);
   271→        break;
   272→      case "H":
   273→        setH(v);
   274→        break;
   275→      case "L":
   276→        setL(v);
   277→        break;
   278→      case "W":
   279→        setW(v);
   280→        break;
   281→      case "S":
   282→        setS(v);
   283→        break;
   284→      case "C":
   285→        setC(v);
   286→        break;
   287→    }
   288→  };
   289→
   290→  /* ---------------------------- rendu ------------------------------ */
   291→  return (
   292→    <div className="flex min-h-screen flex-col">
   293→      {/* En-tête */}
   294→      <header className="sticky top-0 z-40 border-b border-border/70 bg-background/80 backdrop-blur-md">
   295→        <div className="mx-auto flex h-14 w-full max-w-7xl items-center gap-2 px-4 sm:gap-3">
   296→          <button
   297→            onClick={onHome}
   298→            className="flex items-center gap-2.5 rounded-xl outline-none ring-primary/50 transition hover:opacity-85 focus-visible:ring-2"
   299→            aria-label="Retour à l'accueil"
   300→          >
   301→            <span className="grid size-9 place-items-center rounded-xl bg-gradient-to-br from-[#D6336C] to-[#F0703F] text-white shadow-sm">
   302→              <Scissors className="size-[18px]" />
   303→            </span>
   304→            <span className="hidden leading-tight sm:block">
   305→              <span className="block font-display text-[17px] font-bold">
   306→                Atelier de coupe
   307→              </span>
   308→              <span className="block text-[11px] text-muted-foreground">
   309→                Patronage · mise en plan · assemblage
   310→              </span>
   311→            </span>
   312→          </button>
   313→          <Button
   314→            variant="outline"
   315→            size="sm"
   316→            onClick={onHome}
   317→            className="ml-1 gap-1 rounded-full px-3 text-xs"
   318→          >
   319→            <ArrowLeft className="size-3.5" />
   320→            Accueil
   321→          </Button>
   322→          <div className="ml-auto flex items-center gap-2">
   323→            <Badge
   324→              variant="secondary"
   325→              className="hidden gap-1 rounded-full bg-accent px-3 text-[11px] font-semibold text-accent-foreground md:flex"
   326→            >
   327→              <Sparkles className="size-3 text-primary" />
   328→              {Object.keys(MODELS).length} modèles
   329→            </Badge>
   330→            <ThemeToggle />
   331→          </div>
   332→        </div>
   333→      </header>
   334→
   335→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 pb-14 pt-6">
   336→        {/* Introduction */}
   337→        <div className="mb-6">
   338→          <h1 className="font-display text-3xl font-bold leading-tight sm:text-4xl">
   339→            L'atelier,{" "}
   340→            <span className="bg-gradient-to-r from-[#D6336C] to-[#F0703F] bg-clip-text text-transparent dark:from-[#FF6B9A] dark:to-[#FFA94D]">
   341→              du tissu à la pièce finie
   342→            </span>
   343→          </h1>
   344→          <p className="mt-2 max-w-2xl text-sm leading-relaxed text-muted-foreground">
   345→            Choisissez un modèle, entrez les mesures : les pièces sont placées
   346→            sur le tissu avec leurs marges de couture. Touchez une pièce pour la
   347→            couper, suivez la méthode pas à pas, puis assemblez — l'aperçu du
   348→            vêtement se met à jour en temps réel.
   349→          </p>
   350→        </div>
   351→
   352→        <div className="grid grid-cols-1 items-start gap-5 lg:grid-cols-[340px_minmax(0,1fr)]">
   353→          {/* Barre latérale */}
   354→          <aside className="flex flex-col gap-5 lg:sticky lg:top-20">
   355→            <ModelsCard value={modelKey} onChange={changeModel} />
   356→            <MeasuresCard
   357→              P={P}
   358→              T={T}
   359→              H={H}
   360→              L={L}
   361→              W={W}
   362→              S={S}
   363→              C={C}
   364→              set={setMeasure}
   365→            />
   366→            <ClientsCard
   367→              clients={clients}
   368→              ready={clients !== null}
   369→              busy={clBusy}
   370→              selected={clientSel}
   371→              onSelect={applyClient}
   372→              name={clientName}
   373→              onName={setClientName}
   374→              onSave={() => void saveClient()}
   375→              onDelete={() => void deleteClient()}
   376→            />
   377→          </aside>
   378→
   379→          {/* Colonne principale */}
   380→          <div className="flex min-w-0 flex-col gap-6">
   381→            {/* Plan de coupe */}
   382→            <motion.section
   383→              {...reveal}
   384→              aria-label="Plan de coupe"
   385→              className="card-luxe overflow-hidden rounded-2xl border border-border/70 bg-card"
   386→            >
   387→              <header className="flex flex-wrap items-center gap-3 border-b border-border/60 px-5 py-4">
   388→                <div className="grid size-9 shrink-0 place-items-center rounded-lg bg-accent text-primary">
   389→                  <ModelIcon kind={modelKey} className="size-5" />
   390→                </div>
   391→                <div className="min-w-0">
   392→                  <h2 className="font-display text-xl font-bold leading-tight">
   393→                    {model.n}
   394→                  </h2>
   395→                  <p className="flex items-center gap-1 text-xs text-muted-foreground">
   396→                    <MousePointerClick className="size-3" />
   397→                    Touchez une pièce pour la couper
   398→                  </p>
   399→                </div>
   400→                <div className="ml-auto flex items-center gap-2">
   401→                  <Badge
   402→                    variant="outline"
   403→                    className="gap-1.5 rounded-full border-border/80 bg-background font-normal"
   404→                  >
   405→                    <Ruler className="size-3.5 text-primary" />
   406→                    <span className="hidden sm:inline">Métrage</span>{" "}
   407→                    <b className="tabular-nums">{meters} m</b>
   408→                  </Badge>
   409→                  <Badge
   410→                    variant="outline"
   411→                    className="hidden gap-1.5 rounded-full border-border/80 bg-background font-normal md:inline-flex"
   412→                  >
   413→                    <MoveHorizontal className="size-3.5 text-primary" />
   414→                    <b className="tabular-nums">{layout.raw} cm</b>
   415→                  </Badge>
   416→                  <ProgressRing value={cutCount} total={total} size={44} />
   417→                </div>
   418→              </header>
   419→
   420→              <div className="flex flex-col gap-3 p-4 sm:p-5">
   421→                {tooNarrow && (
   422→                  <div className="flex items-start gap-2.5 rounded-xl border border-destructive/30 bg-destructive/10 px-4 py-3 text-sm">
   423→                    <TriangleAlert className="mt-0.5 size-4 shrink-0 text-destructive" />
   424→                    <span>
   425→                      Le tissu est trop étroit. Il faut au moins{" "}
   426→                      <b>{Math.ceil(layout.mx)} cm</b> de largeur pour ce
   427→                      modèle.
   428→                    </span>
   429→                  </div>
   430→                )}
   431→
   432→                <div className="flex flex-wrap items-center gap-2">
   433→                  <Button
   434→                    onClick={cutAll}
   435→                    disabled={cutCount === total}
   436→                    className="rounded-lg"
   437→                  >
   438→                    <Scissors className="size-4" />
   439→                    Tout couper
   440→                  </Button>
   441→                  <Button
   442→                    variant="outline"
   443→                    onClick={resetCut}
   444→                    className="rounded-lg"
   445→                  >
   446→                    <RotateCcw className="size-4" />
   447→                    Recommencer
   448→                  </Button>
   449→                  <Button
   450→                    variant="outline"
   451→                    onClick={() => setZoom((z) => (z === 1 ? 2 : 1))}
   452→                    className="rounded-lg"
   453→                  >
   454→                    {zoom === 1 ? (
   455→                      <ZoomIn className="size-4" />
   456→                    ) : (
   457→                      <ZoomOut className="size-4" />
   458→                    )}
   459→                    {zoom === 1 ? "Zoom ×2" : "Zoom ×1"}
   460→                  </Button>
   461→                  <p className="ml-auto hidden text-xs tabular-nums text-muted-foreground md:block">
   462→                    {cutCount} pièce{cutCount > 1 ? "s" : ""} coupée
   463→                    {cutCount > 1 ? "s" : ""} sur {total}
   464→                  </p>
   465→                </div>
   466→
   467→                <FabricTable
   468→                  layout={layout}
   469→                  fc={C}
   470→                  sa={sa}
   471→                  cut={cut}
   472→                  cuttingId={cuttingId}
   473→                  onCut={(id) => void runCut(id)}
   474→                  zoom={zoom}
   475→                />
   476→
   477→                <p className="text-xs leading-relaxed text-muted-foreground">
   478→                  Marges de couture de {sa} cm comprises autour de chaque pièce
   479→                  · lisières en pointillés · règle graduée en cm · flèche =
   480→                  droit-fil · « pli » = à couper au pli.
   481→                </p>
   482→              </div>
   483→            </motion.section>
   484→
   485→            {/* Aperçu & style */}
   486→            <motion.section
   487→              {...reveal}
   488→              aria-label="Aperçu du vêtement et du style"
   489→              className="card-luxe overflow-hidden rounded-2xl border border-border/70 bg-card"
   490→            >
   491→              <header className="flex items-center gap-3 border-b border-border/60 px-5 py-4">
   492→                <div className="grid size-9 shrink-0 place-items-center rounded-lg bg-accent text-primary">
   493→                  <Eye className="size-[18px]" />
   494→                </div>
   495→                <div className="min-w-0">
   496→                  <h2 className="font-display text-[17px] font-bold leading-tight">
   497→                    Aperçu & style
   498→                  </h2>
   499→                  <p className="text-xs text-muted-foreground">
   500→                    Le vêtement obtenu, dessiné sur vos mesures
   501→                  </p>
   502→                </div>
   503→                <Badge
   504→                  variant="outline"
   505→                  className="ml-auto hidden gap-1 rounded-full border-border/80 bg-background font-normal sm:inline-flex"
   506→                >
   507→                  <Sparkles className="size-3.5 text-primary" />
   508→                  Temps réel
   509→                </Badge>
   510→              </header>
   511→              <div className="grid items-center gap-5 p-5 md:grid-cols-[minmax(0,260px)_minmax(0,1fr)]">
   512→                <div className="rounded-xl border border-border/60 bg-gradient-to-b from-accent/60 via-background to-background p-4">
   513→                  <GarmentPreview
   514→                    m={mm}
   515→                    modelKey={modelKey}
   516→                    fc={C}
   517→                    className="mx-auto h-64 w-auto max-w-full sm:h-72"
   518→                    label={`Aperçu : ${model.n}`}
   519→                  />
   520→                </div>
   521→                <div className="min-w-0">
   522→                  <h3 className="font-display text-2xl font-bold leading-tight">
   523→                    {model.n}
   524→                  </h3>
   525→                  <p className="mt-1.5 text-sm leading-relaxed text-muted-foreground">
   526→                    {model.desc}
   527→                  </p>
   528→                  <div className="mt-3 flex flex-wrap gap-1.5">
   529→                    {model.tags.map((t) => (
   530→                      <Badge
   531→                        key={t}
   532→                        variant="secondary"
   533→                        className="rounded-full bg-accent px-2.5 text-[11px] font-medium text-accent-foreground"
   534→                      >
   535→                        {t}
   536→                      </Badge>
   537→                    ))}
   538→                  </div>
   539→                  <dl className="mt-4 grid grid-cols-2 gap-x-4 gap-y-3 text-sm sm:grid-cols-3">
   540→                    <div>
   541→                      <dt className="text-[11px] uppercase tracking-wider text-muted-foreground">
   542→                        Difficulté
   543→                      </dt>
   544→                      <dd className="mt-1 flex items-center gap-1">
   545→                        {[1, 2, 3].map((i) => (
   546→                          <span
   547→                            key={i}
   548→                            className={`size-2 rounded-full ${
   549→                              i <= model.diff ? "bg-primary" : "bg-border"
   550→                            }`}
   551→                          />
   552→                        ))}
