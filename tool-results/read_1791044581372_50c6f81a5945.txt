     1→/**
     2→ * Bibliothèque de patronage — Atelier de coupe
     3→ * Génération des pièces (chemins SVG en centimètres), placement sur le tissu,
     4→ * méthode de coupe pas à pas et séquences d'assemblage.
     5→ * 17 modèles paramétriques calculés sur les mesures P·T·H·L.
     6→ */
     7→
     8→export interface PieceDef {
     9→  /** nom de la pièce */
    10→  n: string;
    11→  /** chemin SVG (unités = cm) */
    12→  d: string;
    13→  w: number;
    14→  h: number;
    15→  /** nombre d'exemplaires à couper */
    16→  q: number;
    17→  /** pièce à couper au pli */
    18→  fold?: boolean;
    19→  /** remplissage even-odd (trou central, jupe mouchoir) */
    20→  eo?: boolean;
    21→  note?: string;
    22→}
    23→
    24→export interface Measures {
    25→  P: number;
    26→  T: number;
    27→  H: number;
    28→  L: number;
    29→}
    30→

    68→export interface Model {
    69→  n: string;
    70→  L: number;
    71→  cat: CatKey;
    72→  desc: string;
    73→  tags: string[];
    74→  diff: 1 | 2 | 3;
    75→  fab: string;
    76→  g: (m: Measures) => PieceDef[];

    79→/* ------------------------------------------------------------------ */
    80→/* Briques de patronage                                                */
    81→/* ------------------------------------------------------------------ */
    82→
    83→const rect = (w: number, h: number) => `M0 0H${w}V${h}H0Z`;
    84→
    85→const belt = (m: Measures, ext = 4): PieceDef => ({
    86→  n: "Ceinture",
    87→  d: rect(m.T + ext, 8),
    88→  w: m.T + ext,
    89→  h: 8,
    90→  q: 1,
    91→});
    92→
    93→const body = (m: Measures, back: boolean, flare: number): PieceDef => {
    94→  const cw = (m.P + 8) / 4;
    95→  const sh = m.P * 0.21;
    96→  const nw = 7.5;
    97→  const nd = back ? 2.5 : 9;
    98→  const ad = m.P / 6 + 9;
    99→  const hm = cw + (m.L - ad) * flare;
   100→  return {
   101→    n: back ? "Dos" : "Devant",
   102→    d: `M0 ${nd}Q${nw * 0.9} ${nd} ${nw} 0L${sh} 2.5Q${sh - 2} ${ad * 0.6} ${cw} ${ad}L${hm} ${m.L}H0Z`,
   103→    w: Math.max(hm, cw),
   104→    h: m.L,
   105→    q: 1,
   106→    fold: true,
   107→  };
   108→};
   109→
   110→const sleeve = (m: Measures, ml: number, fl = 0): PieceDef => {
   111→  const W = m.P * 0.3 + 8;
   112→  const cap = m.P / 9;
   113→  const hb = Math.max(2, 3 - fl);
   114→  return {
   115→    n: "Manche",
   116→    d: `M0 ${cap}Q${W * 0.15} ${cap * 0.1} ${W / 2} 0Q${W * 0.85} ${cap * 0.1} ${W} ${cap}L${W - 3 + fl} ${ml}H${hb}Z`,
   117→    w: Math.max(W, W - 3 + fl),
   118→    h: ml,
   119→    q: 2,
   120→  };
   121→};
   122→
   123→/** Panneau de jupe : courbe de hanches puis évasement (flare négatif = crayon) */
   124→const skirtPanel = (m: Measures, back: boolean, flare: number): PieceDef => {
   125→  const hw = (m.H + 4) / 4 + (back ? 1 : 0);
   126→  const tw = (m.T + 4) / 4;
   127→  const hipY = Math.min(22, m.L * 0.35);
   128→  const hem = hw + (m.L - hipY) * flare;
   129→  return {
   130→    n: back ? "Dos" : "Devant",
   131→    d: `M0 0H${tw}C${tw + (hw - tw) * 0.25} ${hipY * 0.4} ${hw - (hw - tw) * 0.1} ${hipY * 0.75} ${hw} ${hipY}L${hem} ${m.L}H0Z`,
   132→    w: Math.max(hem, hw),
   133→    h: m.L,
   134→    q: 1,
   135→    fold: true,
   136→  };
   137→};
   138→
   139→/** Jambe de pantalon / short (wide = évasement de l'ourlet) */
   140→const leg = (m: Measures, b: boolean, wide = 0): PieceDef => {
   141→  const hw = (m.H + 4) / 4 + (b ? 2 : 0);
   142→  const c = m.H / 16 + (b ? 2 : 0);
   143→  const cd = Math.min(26, m.L * 0.7);
   144→  const sp = wide * Math.max(0, m.L * 0.16);
   145→  return {
   146→    n: b ? "Dos" : "Devant",
   147→    d: `M${c} 0H${c + hw - 1.5}L${c + hw} ${cd * 0.8}L${c + hw + sp} ${m.L}H0V${cd}Q${c - 1} ${cd * 0.85} ${c} 0Z`,
   148→    w: c + hw + sp,
   149→    h: m.L,
   150→    q: 2,
   151→  };
   152→};
   153→
   154→/** Panneau plissé : la profondeur des plis est comprise dans la largeur */
   155→const pleatPanel = (m: Measures, back: boolean): PieceDef => {
   156→  const w = ((m.T + 6) / 2) * 2.2;
   157→  return {
   158→    n: back ? "Dos plissé" : "Devant plissé",
   159→    d: rect(w, m.L),
   160→    w,
   161→    h: m.L,
   162→    q: 1,
   163→    fold: true,
   164→    note: "plis compris",
   165→  };
   166→};
   167→
   168→/** Devant croisé de la jupe portefeuille */
   169→const wrapFront = (m: Measures): PieceDef => {
   170→  const hw = (m.H + 4) / 4;
   171→  const tw = (m.T + 4) / 4;
   172→  const w = hw * 1.45;
   173→  return {
   174→    n: "Devant",
   175→    d: `M0 0H${tw}C${tw + (hw - tw) * 0.25} 8 ${hw} 14 ${hw} 22L${w} ${m.L}L0 ${m.L}Z`,
   176→    w,
   177→    h: m.L,
   178→    q: 1,
   179→    fold: true,
   180→    note: "panneau croisé",
   181→  };
   182→};
   183→
   184→/** Dos kimono : pièce en T (dos + manches droites d'une seule tenue) */
   185→const kimonoBack = (m: Measures): PieceDef => {
   186→  const nw = 8;
   187→  const arm = m.P * 0.24;
   188→  const sd = Math.max(40, m.P / 4);
   189→  const cw = (m.P + 10) / 4;
   190→  const w = nw + arm + 6;
   191→  return {
   192→    n: "Dos",
   193→    d: `M0 0H${nw}L${w} 0V${sd}H${cw}V${m.L}H0Z`,
   194→    w,
   195→    h: m.L,
   196→    q: 1,
   197→    fold: true,
   198→    note: "pièce en T",
   199→  };
   200→};
   201→
   202→const kimonoFront = (m: Measures): PieceDef => {
   203→  const nw = 8;
   204→  const arm = m.P * 0.24;
   205→  const sd = Math.max(40, m.P / 4);
   206→  const cw = (m.P + 10) / 4;
   207→  const w = nw + arm + 6;
   208→  return {
   209→    n: "Devant",
   210→    d: `M0 10L${nw} 0L${w} 0V${sd}H${cw}V${m.L}H0Z`,
   211→    w,
   212→    h: m.L,
   213→    q: 2,
   214→    note: "croisé",
   215→  };
   216→};

   505→/* ------------------------------------------------------------------ */
   506→/* Placement sur le tissu (algorithme de rayonnage)                    */
   507→/* ------------------------------------------------------------------ */
   508→
   509→export interface PlacedPiece extends PieceDef {
   510→  id: number;
   511→  name: string;
   512→  /** second exemplaire d'une paire = miroir */
   513→  mir: boolean;
   514→  x: number;
   515→  y: number;
   516→}
   517→
   518→export interface BuiltLayout {
   519→  pieces: PlacedPiece[];
   520→  /** largeur de tissu retenue (cm) */
   521→  Wf: number;
   522→  /** hauteur (métrage) du plan (cm) */
   523→  H: number;
   524→  /** plus grande largeur de pièce, marges comprises (cm) */
   525→  mx: number;
   526→  /** largeur demandée (cm) */
   527→  raw: number;
   528→}
   529→
   530→export function buildLayout(
   531→  defs: PieceDef[],
   532→  sa: number,
   533→  fabricWidth: number
   534→): BuiltLayout {
   535→  const out: PlacedPiece[] = [];
   536→  defs.forEach((p) => {
   537→    for (let k = 0; k < p.q; k++) {
   538→      out.push({
   539→        ...p,
   540→        id: out.length,
   541→        name: p.q > 1 ? `${p.n} ${k + 1}/${p.q}` : p.n,
   542→        mir: p.q > 1 && k === 1,
   543→        x: 0,
   544→        y: 0,
   545→      });
   546→    }
   547→  });
   548→  out.sort((a, b) => b.h - a.h);
   549→
   550→  const w0 = fabricWidth || 140;
   551→  let x = 0;
   552→  let y = 0;
   553→  let rh = 0;
   554→  let mx = 0;
   555→
   556→  out.forEach((p) => {
   557→    const bw = p.w + 2 * sa;
   558→    const bh = p.h + 2 * sa;
   559→    mx = Math.max(mx, bw);
   560→    if (x + bw > w0 && x > 0) {
   561→      x = 0;
   562→      y += rh + 1;
   563→      rh = 0;
   564→    }
   565→    p.x = x + sa;
   566→    p.y = y + sa;
   567→    x += bw + 1;
   568→    rh = Math.max(rh, bh);
   569→  });
   570→
   571→  return {
   572→    pieces: out,
   573→    Wf: Math.max(w0, Math.ceil(mx)),
   574→    H: Math.ceil(y + rh),
   575→    mx,
   576→    raw: w0,
   577→  };
   578→}
   579→
   580→/** Métrage nécessaire en mètres (avec 10 cm de marge) */
   581→export const meterage = (layout: BuiltLayout) =>
   582→  (layout.H / 100 + 0.1).toFixed(2);
