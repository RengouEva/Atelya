    55→    55→
    56→    56→export function AtelierApp({
    57→    57→  initialModel,
    58→    58→  onHome,
    59→    59→}: {
    60→    60→  initialModel?: ModelKey;
    61→    61→  onHome: () => void;
    62→    62→}) {
    63→    63→  /* ----------------------------- état ----------------------------- */
    64→    64→  const [modelKey, setModelKey] = React.useState<ModelKey>(initialModel ?? "droite");
    65→    65→  const [P, setP] = React.useState("90");
    66→    66→  const [T, setT] = React.useState("70");
    67→    67→  const [H, setH] = React.useState("98");
    68→    68→  const [L, setL] = React.useState(String(MODELS[initialModel ?? "droite"].L));
    69→    69→  const [W, setW] = React.useState("140");
    70→    70→  const [S, setS] = React.useState("1.5");
    71→    71→  const [C, setC] = React.useState("#2A9DB5");
    72→    72→
    73→    73→  const [zoom, setZoom] = React.useState(1);
    74→    74→  const [cut, setCut] = React.useState<Set<number>>(new Set());
    75→    75→  const [cuttingId, setCuttingId] = React.useState<number | null>(null);
    76→    76→  const [asmStep, setAsmStep] = React.useState(0);
    77→    77→
    78→    78→  const [clients, setClients] = React.useState<ApiClient[] | null>(null);
    79→    79→  const [clBusy, setClBusy] = React.useState(false);
    80→    80→  const [clientSel, setClientSel] = React.useState("");
    81→    81→  const [clientName, setClientName] = React.useState("");
    82→    82→
    83→    83→  const cutRef = React.useRef<Set<number>>(new Set());
    84→    84→  const busyRef = React.useRef<number | false>(false);
    85→    85→  const genRef = React.useRef(0);
    86→    86→
    87→    87→  /* --------------------------- dérivés ---------------------------- */
    88→    88→  const model = MODELS[modelKey];
    89→    89→  const sa = +S || 0;
    90→    90→  const fw = +W || 140;
    91→    91→
    92→    92→  const mm = React.useMemo(
    93→    93→    () => ({
    94→    94→      P: clampN(P, 90, 60, 160),
    95→    95→      T: clampN(T, 70, 40, 160),
    96→    96→      H: clampN(H, 98, 60, 180),
    97→    97→      L: clampN(L, model.L, 15, 200),
    98→    98→    }),
    99→    99→    [P, T, H, L, model.L]
   100→   100→  );
   101→   101→
   102→   102→  const defs = React.useMemo(() => model.g(mm), [model, mm]);
   103→   103→  const layout = React.useMemo(() => buildLayout(defs, sa, fw), [defs, sa, fw]);
   104→   104→  const meters = meterage(layout);
   105→   105→  const tooNarrow = layout.mx > layout.raw;
   106→   106→  const total = layout.pieces.length;
   107→   107→  const cutCount = cut.size;
   108→   108→
   109→   109→  const sig = `${modelKey}|${P}|${T}|${H}|${L}|${W}|${S}|${C}`;
   110→   110→
   111→   111→  /* Toute modification réinitialise la session de coupe */
   112→   112→  React.useEffect(() => {
   113→   113→    genRef.current += 1;
   114→   114→    cutRef.current = new Set();
   115→   115→    setCut(new Set());
   116→   116→    setCuttingId(null);
   117→   117→    busyRef.current = false;
   118→   118→    setAsmStep(0);
   119→   119→  }, [sig]);
   120→   120→
   121→   121→  /* --------------------- fiches clientes (API) --------------------- */
   122→   122→  const refreshClients = React.useCallback(async () => {
   123→   123→    try {
   124→   124→      const r = await fetch("/api/clients", { cache: "no-store" });
   125→   125→      if (!r.ok) throw new Error();
   126→   126→      const j = (await r.json()) as { clients?: ApiClient[] };
   127→   127→      setClients(j.clients ?? []);
   128→   128→    } catch {
   129→   129→      setClients([]);
   130→   130→      toast.error("Backend injoignable : fiches clientes indisponibles.");
   131→   131→    }
   132→   132→  }, []);
   133→   133→
   134→   134→  React.useEffect(() => {
   135→   135→    void refreshClients();
   136→   136→  }, [refreshClients]);
   137→   137→
   138→   138→  const saveClient = async () => {
   139→   139→    const nom = clientName.trim();
   140→   140→    if (!nom) {
   141→   141→      toast.error("Donnez un nom à la fiche avant d'enregistrer.");
   142→   142→      return;
   143→   143→    }
   144→   144→    setClBusy(true);
   145→   145→    try {
   146→   146→      const payload = { name: nom, P: mm.P, T: mm.T, H: mm.H, L: mm.L };
   147→   147→      const res = await fetch(
   148→   148→        clientSel ? `/api/clients/${clientSel}` : "/api/clients",
   149→   149→        {
   150→   150→          method: clientSel ? "PUT" : "POST",
   151→   151→          headers: { "Content-Type": "application/json" },
   152→   152→          body: JSON.stringify(payload),
   153→   153→        }
   154→   154→      );
   155→   155→      const j = (await res.json().catch(() => ({}))) as {
   156→   156→        client?: ApiClient;
   157→   157→        error?: string;
   158→   158→      };
   159→   159→      if (!res.ok || !j.client) throw new Error(j.error ?? "Enregistrement impossible.");
   160→   160→      const client = j.client;
   161→   161→      setClients((prev) =>
   162→   162→        prev
   163→   163→          ? clientSel
   164→   164→            ? prev.map((c) => (c.id === client.id ? client : c))
   165→   165→            : [...prev, client]
   166→   166→          : [client]
   167→   167→      );
   168→   168→      setClientSel(client.id);
   169→   169→      setClientName("");
   170→   170→      toast.success(
   171→   171→        clientSel ? `Fiche de ${nom} mise à jour.` : `Fiche de ${nom} enregistrée.`,
   172→   172→        { description: "Mesures conservées dans la base de l'atelier." }
   173→   173→      );
   174→   174→    } catch (e) {
   175→   175→      toast.error(e instanceof Error ? e.message : "Enregistrement impossible.");
   176→   176→    } finally {
   177→   177→      setClBusy(false);
   178→   178→    }
   179→   179→  };
   180→   180→
   181→   181→  const deleteClient = async () => {
   182→   182→    if (!clientSel || clBusy) return;
   183→   183→    setClBusy(true);
   184→   184→    const c = clients?.find((x) => x.id === clientSel);
   185→   185→    try {
   186→   186→      const res = await fetch(`/api/clients/${clientSel}`, { method: "DELETE" });
   187→   187→      if (!res.ok) throw new Error("Suppression impossible.");
   188→   188→      setClients((prev) => prev?.filter((x) => x.id !== clientSel) ?? []);
   189→   189→      setClientSel("");
   190→   190→      if (c) toast(`Fiche de ${c.name} supprimée.`);
   191→   191→    } catch (e) {
   192→   192→      toast.error(e instanceof Error ? e.message : "Suppression impossible.");
   193→   193→    } finally {
   194→   194→      setClBusy(false);
   195→   195→    }
   196→   196→  };
   197→   197→
   198→   198→  const applyClient = (id: string) => {
   199→   199→    setClientSel(id);
   200→   200→    const c = clients?.find((x) => x.id === id);
   201→   201→    if (!c) return;
   202→   202→    setP(String(c.P));
   203→   203→    setT(String(c.T));
   204→   204→    setH(String(c.H));
   205→   205→    setL(String(c.L));
   206→   206→    setClientName(c.name);
   207→   207→    toast(`Mesures de ${c.name} appliquées.`);
   208→   208→  };
   209→   209→
   210→   210→  /* ------------------------- découpe ------------------------------ */
   211→   211→  const runCut = (id: number) => {
   212→   212→    const gen = genRef.current;
   213→   213→    return new Promise<void>((resolve) => {
   214→   214→      if (cutRef.current.has(id) || busyRef.current !== false) {
   215→   215→        resolve();
   216→   216→        return;
   217→   217→      }
   218→   218→      busyRef.current = id;
   219→   219→      setCuttingId(id);
   220→   220→      window.setTimeout(() => {
   221→   221→        if (gen !== genRef.current) {
   222→   222→          resolve();
   223→   223→          return;
   224→   224→        }
   225→   225→        const next = new Set(cutRef.current);
   226→   226→        next.add(id);
   227→   227→        cutRef.current = next;
   228→   228→        setCut(next);
   229→   229→        setCuttingId(null);
   230→   230→        busyRef.current = false;
   231→   231→        if (total > 0 && next.size === total) {
   232→   232→          toast.success("Toutes les pièces sont coupées !", {
   233→   233→            description:
   234→   234→              "Reportez les repères avant de retirer le patron, puis passez à l'assemblage.",
   235→   235→          });
   236→   236→        }
   237→   237→        resolve();
   238→   238→      }, 1050);
   239→   239→    });
   240→   240→  };
   241→   241→
   242→   242→  const cutAll = async () => {
   243→   243→    if (busyRef.current !== false) return;
   244→   244→    for (const p of layout.pieces) {
   245→   245→      if (cutRef.current.has(p.id)) continue;
   246→   246→      await runCut(p.id);
   247→   247→    }
   248→   248→  };
   249→   249→
   250→   250→  const resetCut = () => {
   251→   251→    genRef.current += 1;
   252→   252→    cutRef.current = new Set();
   253→   253→    setCut(new Set());
   254→   254→    setCuttingId(null);
   255→   255→    busyRef.current = false;
   256→   256→  };
   257→   257→
   258→   258→  /* --------------------------- divers ------------------------------ */
   259→   259→  const changeModel = (k: ModelKey) => {
   260→   260→    setModelKey(k);
   261→   261→    setL(String(MODELS[k].L));
   262→   262→  };
   263→   263→
   264→   264→  const setMeasure = (k: MeasureField, v: string) => {
   265→   265→    switch (k) {
   266→   266→      case "P":
   267→   267→        setP(v);
   268→   268→        break;
   269→   269→      case "T":
   270→   270→        setT(v);
   271→   271→        break;
   272→   272→      case "H":
   273→   273→        setH(v);
   274→   274→        break;
   275→   275→      case "L":
   276→   276→        setL(v);
   277→   277→        break;
   278→   278→      case "W":
   279→   279→        setW(v);
   280→   280→        break;
   281→   281→      case "S":
   282→   282→        setS(v);
   283→   283→        break;
   284→   284→      case "C":
   285→   285→        setC(v);
   286→   286→        break;
   287→   287→    }
   288→   288→  };
   289→   289→
   290→   290→  /* ---------------------------- rendu ------------------------------ */
   291→   291→  return (
   292→   292→    <div className="flex min-h-screen flex-col">
   293→   293→      {/* En-tête */}
   294→   294→      <header className="sticky top-0 z-40 border-b border-border/70 bg-background/80 backdrop-blur-md">
   295→   295→        <div className="mx-auto flex h-14 w-full max-w-7xl items-center gap-2 px-4 sm:gap-3">
   296→   296→          <button
   297→   297→            onClick={onHome}
   298→   298→            className="flex items-center gap-2.5 rounded-xl outline-none ring-primary/50 transition hover:opacity-85 focus-visible:ring-2"
   299→   299→            aria-label="Retour à l'accueil"
   300→   300→          >
   301→   301→            <span className="grid size-9 place-items-center rounded-xl bg-gradient-to-br from-[#D6336C] to-[#F0703F] text-white shadow-sm">
   302→   302→              <Scissors className="size-[18px]" />
   303→   303→            </span>
   304→   304→            <span className="hidden leading-tight sm:block">
   305→   305→              <span className="block font-display text-[17px] font-bold">
   306→   306→                Atelier de coupe
   307→   307→              </span>
   308→   308→              <span className="block text-[11px] text-muted-foreground">
   309→   309→                Patronage · mise en plan · assemblage
   310→   310→              </span>
   311→   311→            </span>
   312→   312→          </button>
   313→   313→          <Button
   314→   314→            variant="outline"
   315→   315→            size="sm"
   316→   316→            onClick={onHome}
   317→   317→            className="ml-1 gap-1 rounded-full px-3 text-xs"
   318→   318→          >
   319→   319→            <ArrowLeft className="size-3.5" />
   320→   320→            Accueil
   321→   321→          </Button>
   322→   322→          <div className="ml-auto flex items-center gap-2">
   323→   323→            <Badge
   324→   324→              variant="secondary"
   325→   325→              className="hidden gap-1 rounded-full bg-accent px-3 text-[11px] font-semibold text-accent-foreground md:flex"
   326→   326→            >
   327→   327→              <Sparkles className="size-3 text-primary" />
   328→   328→              {Object.keys(MODELS).length} modèles
   329→   329→            </Badge>
   330→   330→            <ThemeToggle />
   331→   331→          </div>
   332→   332→        </div>
   333→   333→      </header>
   334→   334→
   335→   335→      <main className="mx-auto w-full max-w-7xl flex-1 px-4 pb-14 pt-6">
   336→   336→        {/* Introduction */}
   337→   337→        <div className="mb-6">
   338→   338→          <h1 className="font-display text-3xl font-bold leading-tight sm:text-4xl">
   339→   339→            L'atelier,{" "}
   340→   340→            <span className="bg-gradient-to-r from-[#D6336C] to-[#F0703F] bg-clip-text text-transparent dark:from-[#FF6B9A] dark:to-[#FFA94D]">
   341→   341→              du tissu à la pièce finie
   342→   342→            </span>
   343→   343→          </h1>
   344→   344→          <p className="mt-2 max-w-2xl text-sm leading-relaxed text-muted-foreground">
   345→   345→            Choisissez un modèle, entrez les mesures : les pièces sont placées
   346→   346→            sur le tissu avec leurs marges de couture. Touchez une pièce pour la
   347→   347→            couper, suivez la méthode pas à pas, puis assemblez — l'aperçu du
   348→   348→            vêtement se met à jour en temps réel.
   349→   349→          </p>
   350→   350→        </div>
   351→   351→
   352→   352→        <div className="grid grid-cols-1 items-start gap-5 lg:grid-cols-[340px_minmax(0,1fr)]">
   353→   353→          {/* Barre latérale */}
   354→   354→          <aside className="flex flex-col gap-5 lg:sticky lg:top-20">
   355→   355→            <ModelsCard value={modelKey} onChange={changeModel} />
   356→   356→            <MeasuresCard
   357→   357→              P={P}
   358→   358→              T={T}
   359→   359→              H={H}
   360→   360→              L={L}
   361→   361→              W={W}
   362→   362→              S={S}
   363→   363→              C={C}
   364→   364→              set={setMeasure}
   365→   365→            />
   366→   366→            <ClientsCard
   367→   367→              clients={clients}
   368→   368→              ready={clients !== null}
   369→   369→              busy={clBusy}
   370→   370→              selected={clientSel}
   371→   371→              onSelect={applyClient}
   372→   372→              name={clientName}
   373→   373→              onName={setClientName}
   374→   374→              onSave={() => void saveClient()}
   375→   375→              onDelete={() => void deleteClient()}
   376→   376→            />
   377→   377→          </aside>
   378→   378→
   379→   379→          {/* Colonne principale */}
   380→   380→          <div className="flex min-w-0 flex-col gap-6">
   381→   381→            {/* Plan de coupe */}
   382→   382→            <motion.section
   383→   383→              {...reveal}
   384→   384→              aria-label="Plan de coupe"
   385→   385→              className="card-luxe overflow-hidden rounded-2xl border border-border/70 bg-card"
   386→   386→            >
   387→   387→              <header className="flex flex-wrap items-center gap-3 border-b border-border/60 px-5 py-4">
   388→   388→                <div className="grid size-9 shrink-0 place-items-center rounded-lg bg-accent text-primary">
   389→   389→                  <ModelIcon kind={modelKey} className="size-5" />
   390→   390→                </div>
   391→   391→                <div className="min-w-0">
   392→   392→                  <h2 className="font-display text-xl font-bold leading-tight">
   393→   393→                    {model.n}
   394→   394→                  </h2>
   395→   395→                  <p className="flex items-center gap-1 text-xs text-muted-foreground">
   396→   396→                    <MousePointerClick className="size-3" />
   397→   397→                    Touchez une pièce pour la couper
   398→   398→                  </p>
   399→   399→                </div>
   400→   400→                <div className="ml-auto flex items-center gap-2">
   401→   401→                  <Badge
   402→   402→                    variant="outline"
   403→   403→                    className="gap-1.5 rounded-full border-border/80 bg-background font-normal"
   404→   404→                  >
   405→   405→                    <Ruler className="size-3.5 text-primary" />
   406→   406→                    <span className="hidden sm:inline">Métrage</span>{" "}
   407→   407→                    <b className="tabular-nums">{meters} m</b>
   408→   408→                  </Badge>
   409→   409→                  <Badge
   410→   410→                    variant="outline"
   411→   411→                    className="hidden gap-1.5 rounded-full border-border/80 bg-background font-normal md:inline-flex"
   412→   412→                  >
   413→   413→                    <MoveHorizontal className="size-3.5 text-primary" />
   414→   414→                    <b className="tabular-nums">{layout.raw} cm</b>
   415→   415→                  </Badge>
   416→   416→                  <ProgressRing value={cutCount} total={total} size={44} />
   417→   417→                </div>
   418→   418→              </header>
   419→   419→
   420→   420→              <div className="flex flex-col gap-3 p-4 sm:p-5">
   421→   421→                {tooNarrow && (
   422→   422→                  <div className="flex items-start gap-2.5 rounded-xl border border-destructive/30 bg-destructive/10 px-4 py-3 text-sm">
   423→   423→                    <TriangleAlert className="mt-0.5 size-4 shrink-0 text-destructive" />
   424→   424→                    <span>
   425→   425→                      Le tissu est trop étroit. Il faut au moins{" "}
   426→   426→                      <b>{Math.ceil(layout.mx)} cm</b> de largeur pour ce
   427→   427→                      modèle.
   428→   428→                    </span>
   429→   429→                  </div>
   430→   430→                )}
   431→   431→
   432→   432→                <div className="flex flex-wrap items-center gap-2">
   433→   433→                  <Button
   434→   434→                    onClick={cutAll}
   435→   435→                    disabled={cutCount === total}
   436→   436→                    className="rounded-lg"
   437→   437→                  >
   438→   438→                    <Scissors className="size-4" />
   439→   439→                    Tout couper
   440→   440→                  </Button>
   441→   441→                  <Button
   442→   442→                    variant="outline"
   443→   443→                    onClick={resetCut}
   444→   444→                    className="rounded-lg"
   445→   445→                  >
   446→   446→                    <RotateCcw className="size-4" />
   447→   447→                    Recommencer
   448→   448→                  </Button>
   449→   449→                  <Button
   450→   450→                    variant="outline"
   451→   451→                    onClick={() => setZoom((z) => (z === 1 ? 2 : 1))}
   452→   452→                    className="rounded-lg"
   453→   453→                  >
   454→   454→                    {zoom === 1 ? (
   455→   455→                      <ZoomIn className="size-4" />
   456→   456→                    ) : (
   457→   457→                      <ZoomOut className="size-4" />
   458→   458→                    )}
   459→   459→                    {zoom === 1 ? "Zoom ×2" : "Zoom ×1"}
   460→   460→                  </Button>
   461→   461→                  <p className="ml-auto hidden text-xs tabular-nums text-muted-foreground md:block">
   462→   462→                    {cutCount} pièce{cutCount > 1 ? "s" : ""} coupée
